580 void floodFill4_Stack(bool* visisted, P* screenBuffer, int w, int h, int x, int y, int value, double sigma, std::vector<std::pair<double, double>>& vecOut)
581 {
582 static const int dx[4] = { 0, 1, 0, -1 }; // relative neighbor x coordinates
583 static const int dy[4] = { -1, 0, 1, 0 }; // relative neighbor y coordinates
584
585 std::vector<int> stack;
586
587 push(stack, x, y);
588 while (pop(stack, x, y))
589 {
590 visisted[(h - y) * w + x] = true;
591 screenBuffer[(h - y) * w + x] = 500;
592
593 for (int i = 0; i < 4; i++) {
594 int nx = x + dx[i];
595 int ny = y + dy[i];
596 if (nx >= 0 && nx < w && ny >= 0 && ny < h && (screenBuffer[(h - ny) * w + nx] >= value * (1 - sigma) && !visisted[(h - ny) * w + nx])) {
597 push(stack, nx, ny);
598 vecOut.push_back(std::make_pair(nx, ny));
599 }
600 }
601 }
